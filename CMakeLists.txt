cmake_minimum_required(VERSION 3.20)
project(Methusalah VERSION 0.1.0 LANGUAGES CXX)
option(METHUSALAH_BUILD_CLI "Build the command-line tool" ON)

add_library(methusalah src/format/header.cpp src/format/file_manager.cpp src/inference/engine.cpp)
add_library(Methusalah::methusalah ALIAS methusalah)
target_compile_features(methusalah PUBLIC cxx_std_20)
target_include_directories(methusalah PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
set_target_properties(methusalah PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(methusalah PRIVATE /W4)
else()
    target_compile_options(methusalah PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(METHUSALAH_BUILD_CLI)
    add_executable(methusalah_cli src/main.cpp)
    target_link_libraries(methusalah_cli PRIVATE Methusalah::methusalah)
    set_target_properties(methusalah_cli PROPERTIES OUTPUT_NAME methusalah CXX_EXTENSIONS OFF)
endif()
